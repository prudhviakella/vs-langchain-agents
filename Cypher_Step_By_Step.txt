//================================================================================
// SECTION 1 — YOUR VERY FIRST NODE
//================================================================================

// A NODE is a single thing. A customer. A product. An order.
// A node can have:
//   - one or more LABELS   : what kind of thing it is        (:Customer)
//   - any number of PROPERTIES : facts about it              {name: "Carol"}

// That is the whole idea. Let's make one.


// ---------------------------------------------------------------- 1.1
// The simplest possible node: one label, no properties.
// CREATE makes something new.

CREATE (:Customer);

// --> "Added 1 label, created 1 node".
//     Nothing is displayed, because we did not ask for anything back.


// ---------------------------------------------------------------- 1.2
// To see a node, we have to ask for it. MATCH finds, RETURN displays.
// The "c" is a VARIABLE — a temporary name so we can refer to the node.

MATCH (c:Customer)
RETURN c;

// --> One empty node. Click it in the graph view; it has no properties yet.


// ---------------------------------------------------------------- 1.3
// Delete it and start again, this time with properties.
// DETACH DELETE removes a node and anything attached to it.

MATCH (c:Customer {customerId: 3068})
DETACH DELETE c;


// ---------------------------------------------------------------- 1.4
// A node with properties. Properties go in curly braces {} as key: value.
// This is a real customer from our dataset.

CREATE (c:Customer {
  customerId: 3068,
  firstName:  "Carol",
  lastName:   "Glass",
  city:       "North Royport",
  state:      "NY",
  segment:    "VIP"
})
RETURN c;

// --> One node, six properties. Click it to inspect them.
//     Note we added RETURN c, so this time we see what we made.


// ---------------------------------------------------------------- 1.5
// Properties are TYPED. Cypher supports several types, and the type matters
// because it decides what you can do with the value.

MATCH (c:Customer {customerId: 3068})
SET c.loyaltyPoints = 17064,                    // integer
    c.lifetimeRevenue = 24338.54,               // float  (decimal point)
    c.newsletter = true,                        // boolean (true / false)
    c.joined = datetime("2023-04-11T09:15:00"), // datetime
    c.birthday = date("1988-02-19"),            // date
    c.interests = ["Electronics", "Fitness"]    // list of strings
RETURN c;

// --> Same node, now with eleven properties.
//     Quotes = text. No quotes = number, boolean, or a function call.


// ---------------------------------------------------------------- 1.6
// Return specific properties instead of the whole node.
// Dot notation reads a property: c.firstName

MATCH (c:Customer)
RETURN c.firstName, c.lastName, c.city, c.loyaltyPoints;

// --> A table with four columns instead of a graph picture.


// ---------------------------------------------------------------- 1.7
// Column headings default to the expression, which is ugly.
// AS renames them.

MATCH (c:Customer)
RETURN c.firstName    AS first,
       c.lastName     AS last,
       c.city         AS city,
       c.loyaltyPoints AS points;

// --> Same data, readable headings. Always name your columns.


// ---------------------------------------------------------------- 1.8
// Three useful built-in functions for inspecting a node.

MATCH (c:Customer)
RETURN labels(c)     AS labels,      // which labels it has
       keys(c)       AS propertyKeys, // which properties it has
       properties(c) AS everything;   // all properties as one map

// --> labels: ["Customer"], and a list of your eleven property names.


// ---------------------------------------------------------------- 1.9
// SET also UPDATES an existing property. Same syntax as adding one.

MATCH (c:Customer {customerId: 3068})
SET c.loyaltyPoints = 18000
RETURN c.firstName AS first, c.loyaltyPoints AS points;

// --> points is now 18000. SET overwrote the old value.


// ---------------------------------------------------------------- 1.10
// REMOVE deletes a property entirely. Different from setting it to null —
// though in Neo4j, setting a property to null does remove it.

MATCH (c:Customer {customerId: 3068})
REMOVE c.birthday
RETURN keys(c) AS remainingKeys;

// --> "birthday" is gone from the list. Ten properties left.


// ---------------------------------------------------------------- 1.11
// A node can have MORE THAN ONE label. Labels are how you group nodes.

MATCH (c:Customer {customerId: 3068})
SET c:VIP
RETURN labels(c) AS labels;

// --> ["Customer", "VIP"]. Now this node answers to both.


// ---------------------------------------------------------------- 1.12
// Check it: match on the new label.

MATCH (v:VIP)
RETURN v.firstName AS first, labels(v) AS labels;

// --> Our one node. It is a Customer AND a VIP.


// ---------------------------------------------------------------- 1.13
// Take the extra label off again.

MATCH (c:Customer {customerId: 3068})
REMOVE c:VIP
RETURN labels(c) AS labels;

// --> ["Customer"].


// ---------------------------------------------------------------------------
// CHECKPOINT 1 — you should now be able to answer these without looking:
//   - What is the difference between CREATE and MATCH?
//   - What does RETURN do, and what happens if you leave it out?
//   - How do you read a property off a node?
//   - What is the difference between a label and a property?
// ---------------------------------------------------------------------------


//================================================================================
// SECTION 2 — A SECOND NODE, AND CONNECTING THE TWO
//================================================================================

// A graph is not much use with one node. The point of a graph database is the
// RELATIONSHIPS between things.

// A relationship:
//   - connects exactly two nodes
//   - always has a TYPE, in capitals   -[:BOUGHT]->
//   - always has a DIRECTION           (a)-[:BOUGHT]->(b)
//   - can have properties, just like a node

// We will build this, one step at a time:

//     (:Customer) -[:PLACED]-> (:Order) -[:CONTAINS]-> (:Product)


// ---------------------------------------------------------------- 2.1
// Create a second node — a Product. This is a real product from the dataset.
// Notice: this is completely SEPARATE from our Customer. Nothing links them.

CREATE (p:Product {
  productId: 2732,
  name:      "Flex Sofa",
  brand:     "Horizon",
  price:     1443.44,
  category:  "Treadmills"
})
RETURN p;


// ---------------------------------------------------------------- 2.2
// Look at everything in the database now.
// MATCH (n) with no label means "any node at all".

MATCH (n)
RETURN n;

// --> TWO nodes, floating separately with no line between them.
//     This is the important bit: creating two nodes does NOT connect them.


// ---------------------------------------------------------------- 2.3
// Now connect them. This is the key query of the whole section.
//
// Read it as three steps:
//   1. MATCH the customer, call it c
//   2. MATCH the product,  call it p
//   3. CREATE a relationship from c to p
//
// You must FIND both nodes before you can connect them.


CREATE (c:Customer {
  customerId: 3069,
  firstName:  "Prudhvi",
  lastName:   "Akella",
  city:       "Hyd",
  state:      "TG",
  segment:    "Genral"
})
RETURN c;

MATCH (c:Customer {customerId: 3068})
MATCH (p:Product   {productId:  2732})
CREATE (c)-[:BOUGHT]->(p);


MATCH (c:Customer {customerId: 3068})
MATCH (p:Product   {productId:  2732})
CREATE (c)<-[:BUYERS]-(p);


MATCH (c:Customer {customerId: 3069})
MATCH (p:Product   {productId:  2732})
CREATE (c)-[:BOUGHT]->(p);

MATCH (c:Customer {customerId: 3069})
MATCH (p:Product   {productId:  2732})
CREATE (c)<-[:BUYERS]-(p);

// --> "Created 1 relationship".


// ---------------------------------------------------------------- 2.4
// Look again.

MATCH (n)
RETURN n;

// --> Two nodes with an arrow between them. You have made a graph.


// ---------------------------------------------------------------- 2.5
// Match the pattern explicitly and return both ends.
// The pattern (c)-[:BOUGHT]->(p) is read left to right, following the arrow.

MATCH (c:Customer)-[:BOUGHT]->(p:Product)
RETURN c.firstName AS customer, p.name AS product, p.price AS price;

// --> One row: Carol Glass bought Flex Sofa.


// ---------------------------------------------------------------- 2.6
// DIRECTION MATTERS. This query asks for the arrow the other way round.

MATCH (c:Customer)<-[:BOUGHT]-(p:Product)
RETURN c.firstName AS customer, p.name AS product;

// --> ZERO rows. The relationship goes customer -> product, not product -> customer.
//     This trips up every beginner exactly once.


// ---------------------------------------------------------------- 2.7
// Leave the arrowhead off to match EITHER direction.

MATCH (c:Customer)-[:BOUGHT]-(p:Product)
RETURN c.firstName AS customer, p.name AS product;

// --> One row again. No arrowhead = "connected, don't care which way".


// ---------------------------------------------------------------- 2.8
// Relationships can have properties too. Give the relationship a name (r)
// so you can set things on it.

MATCH (c:Customer)-[r:BOUGHT]->(p:Product)
SET r.quantity = 2,
    r.purchasedOn = date("2024-06-15")
RETURN type(r) AS relType, r.quantity AS qty, r.purchasedOn AS purchased;


MATCH (c:Customer {customerId: 3069})-[r:BOUGHT]->(p:Product)
SET r.quantity = 2,
    r.purchasedOn = date("2024-06-15")
RETURN type(r) AS relType, r.quantity AS qty, r.purchasedOn AS purchased;

// --> type() gives the relationship type as text: "BOUGHT".


// ---------------------------------------------------------------- 2.9
// Our real model puts an Order in between, which is more realistic:
// a customer places an order, and that order contains products.
// First remove the shortcut relationship.

MATCH (:Customer )-[r:BOUGHT]->(:Product)
DELETE r;

// --> "Deleted 1 relationship". The two nodes remain.
//     Note: DELETE for relationships, DETACH DELETE for nodes.


// ---------------------------------------------------------------- 2.10
// Create the Order node and BOTH relationships in one statement.
// You can chain a pattern: (a)-[:X]->(b)-[:Y]->(c)

MATCH (c:Customer {customerId: 3068})
MATCH (p:Product  {productId:  2732})
CREATE (c)-[:PLACED]->(o:Order {
          orderId: 9001,
          orderDate: datetime("2024-06-15T10:30:00"),
          totalAmount: 2886.88,
          status: "Delivered"
       })-[:CONTAINS {quantity: 2, unitPrice: 1443.44}]->(p)
RETURN c, o, p;

// --> Three nodes in a line. Note the CONTAINS relationship carries the
//     quantity and price — facts that belong to the pairing, not to either node.


// ---------------------------------------------------------------- 2.11
// Traverse the whole two-hop path.

MATCH (c:Customer)-[:PLACED]->(o:Order)-[:CONTAINS]->(p:Product)
RETURN c.firstName  AS customer,
       o.orderId    AS orderId,
       o.status     AS status,
       p.name       AS product;

// --> One row spanning three nodes. THIS is a traversal — you followed
//     two relationships to connect a customer to a product.


// ---------------------------------------------------------------- 2.12
// Capture the whole path in a variable with "path ="
// and inspect it with the path functions.

MATCH path = (c:Customer)-[:PLACED]->(:Order)-[:CONTAINS]->(:Product)
RETURN nodes(path)         AS nodesOnPath,
       relationships(path) AS relsOnPath,
       length(path)        AS hops;

// --> hops = 2. length() counts RELATIONSHIPS, not nodes. A 3-node path is 2 hops.


// ---------------------------------------------------------------------------
// CHECKPOINT 2 — you should now be able to answer these:
//   - Why do you need MATCH before CREATE when making a relationship?
//   - What happens if you point the arrow the wrong way?
//   - How do you match a relationship in either direction?
//   - Where do you put a property that belongs to the pairing, not to
//     either node? (Answer: on the relationship.)
//   - Does length(path) count nodes or relationships?
// ---------------------------------------------------------------------------


//================================================================================
// SECTION 3 — BUILDING A SMALL GRAPH
//================================================================================

// Three nodes is still a toy. Now we build up to a real (small) dataset.
// First a few more nodes by hand, so you see the mechanics; then we load the
// full teaching graph in one go.


// ---------------------------------------------------------------- 3.1
// Create several nodes in ONE statement, separated by commas.

CREATE
  (:Product {productId: 2101, name: "Basic Mouse",    brand: "Logitech", price: 24.99}),
  (:Product {productId: 2102, name: "Elite Keyboard", brand: "Logitech", price: 89.50}),
  (:Product {productId: 2103, name: "Pro Headset",    brand: "Sony",     price: 149.00});

// --> "Created 3 nodes". Commas separate independent patterns.


// ---------------------------------------------------------------- 3.2
// Create a Category and link the Logitech products to it.
// One MATCH can return MANY nodes — and then CREATE runs once per match.

CREATE (:Category {categoryId: 900, name: "Computer Accessories", level: 2});

// ---------------------------------------------------------------- 3.3
// This is an important idea: MATCH found 2 products, so CREATE ran twice.

MATCH (p:Product)
WHERE p.brand = "Logitech"
MATCH (cat:Category {categoryId: 900})
CREATE (p)-[:IN_CATEGORY]->(cat);

// --> "Created 2 relationships" — one per matched product.
//     A query runs once for every row that reaches it. Remember this;
//     it explains almost every surprising result you will ever get.


// ---------------------------------------------------------------- 3.4
// Now meet MERGE. MERGE means "find it, or create it if absent".
// Run this query. Then run it AGAIN. Then a third time.

MERGE (b:Brand {name: "Logitech"})
RETURN b, b.name AS brandName;

// --> First run: "Created 1 node". Second and third run: no changes.
//     MERGE is idempotent — safe to re-run. CREATE is not.


// ---------------------------------------------------------------- 3.5
// Compare: run THIS one three times and watch the duplicates pile up.

CREATE (d:Demo {name: "duplicate me"})
RETURN d;

// ---------------------------------------------------------------- 3.6
// Count the damage.

MATCH (d:Demo)
RETURN count(d) AS howManyDemoNodes;

// --> 3 if you ran 3.5 three times. This is why loaders always use MERGE.


// ---------------------------------------------------------------- 3.7
// UNWIND turns a LIST into ROWS. This is how you create many things at once.
// Read it as: "for each map in this list, call it row, then MERGE a node".

UNWIND [
  {id: 2104, name: "Wireless Mouse", brand: "Logitech", price: 39.99},
  {id: 2105, name: "Gaming Chair",   brand: "Secretlab", price: 429.00},
  {id: 2106, name: "USB Hub",        brand: "Anker",    price: 34.95}
] AS row
MERGE (p:Product {productId: row.id})
SET   p.name = row.name, p.brand = row.brand, p.price = row.price
RETURN count(p) AS productsCreated;

// --> 3. UNWIND + MERGE + SET is THE standard loading pattern in Cypher.
//     You will use it for the rest of your life. Learn this shape.


// ---------------------------------------------------------------- 3.8
// See what we have built so far.

MATCH (n)
RETURN labels(n)[0] AS label, count(*) AS count
ORDER BY count DESC;

// --> Product, Category, Customer, Order, Brand, Demo.
//     labels(n)[0] takes the FIRST label, since labels() returns a list.

// ---------------------------------------------------------------- 3.10
// Constraints. A uniqueness constraint does two jobs:
//   1. stops duplicates
//   2. creates an index, which makes MERGE and lookups fast
// ALWAYS create these BEFORE loading data.
// Run all six lines one at a time.

CREATE CONSTRAINT customer_id IF NOT EXISTS FOR (c:Customer) REQUIRE c.customerId IS UNIQUE;

CREATE CONSTRAINT product_id  IF NOT EXISTS FOR (p:Product)  REQUIRE p.productId  IS UNIQUE;

CREATE CONSTRAINT order_id    IF NOT EXISTS FOR (o:Order)    REQUIRE o.orderId    IS UNIQUE;

CREATE CONSTRAINT category_id IF NOT EXISTS FOR (c:Category) REQUIRE c.categoryId IS UNIQUE;

CREATE CONSTRAINT review_id   IF NOT EXISTS FOR (r:Review)   REQUIRE r.reviewId   IS UNIQUE;

CREATE CONSTRAINT supplier_id IF NOT EXISTS FOR (s:Supplier) REQUIRE s.supplierId IS UNIQUE;

// --> Check them with:  SHOW CONSTRAINTS;

//================================================================================
// SECTION 4 — FILTERING WITH WHERE
//================================================================================

// So far we have matched everything. Now we narrow it down.

// There are two places to filter:
//   inline  MATCH (c:Customer {segment: "VIP"})     — only exact equality
//   WHERE   MATCH (c:Customer) WHERE c.segment = "VIP" — anything at all

// Use inline for simple equality, WHERE for everything else.


// ---------------------------------------------------------------- 4.1
// Inline equality — the shortest form.

MATCH (c:Customer {segment: "VIP"})
RETURN c.name AS customer, c.city AS city, c.segment AS segment;

// --> 18 rows.


// ---------------------------------------------------------------- 4.2
// Exactly the same result, written with WHERE.

MATCH (c:Customer)
WHERE c.segment = "VIP"
RETURN c.name AS customer, c.city AS city, c.segment AS segment;

// --> Identical. Note: = is comparison in Cypher, not assignment.


// ---------------------------------------------------------------- 4.3
// Comparison operators:  =  <>  <  >  <=  >=
// <> means "not equal to".

MATCH (p:Product)
WHERE p.price > 1000
RETURN p.name AS product, p.brand AS brand, p.price AS price
ORDER BY p.price DESC;

// --> 35 products over $1000.


// ---------------------------------------------------------------- 4.4
// Combine conditions with AND, OR, NOT. Use parentheses when mixing them.

MATCH (p:Product)
WHERE p.price > 500 AND p.avgRating >= 4.0
RETURN p.name AS product, p.price AS price, p.avgRating AS rating
ORDER BY p.price DESC;

// --> Expensive AND well-rated. Both conditions must hold.


// ---------------------------------------------------------------- 4.5
// OR widens the net.

MATCH (c:Customer)
WHERE c.segment = "Churned" OR c.segment = "At-Risk"
RETURN c.name AS customer, c.segment AS segment, c.city AS city;

// --> 4 rows.


// ---------------------------------------------------------------- 4.6
// IN is much cleaner than a chain of ORs. Same answer as 4.5.

MATCH (c:Customer)
WHERE c.segment IN ["Churned", "At-Risk"]
RETURN c.name AS customer, c.segment AS segment, c.city AS city;

// --> Same 4 rows. Prefer IN when checking a list of values.


// ---------------------------------------------------------------- 4.7
// NOT negates a condition.

MATCH (o:Order)
WHERE NOT o.status = "Delivered"
RETURN o.orderId AS orderId, o.status AS status, o.totalAmount AS amount
ORDER BY o.totalAmount DESC
LIMIT 10;

// --> Everything except Delivered. Same as: WHERE o.status <> "Delivered"


// ---------------------------------------------------------------- 4.8
// Text matching: STARTS WITH, ENDS WITH, CONTAINS. Case-SENSITIVE.

MATCH (p:Product)
WHERE p.name STARTS WITH "Flex"
RETURN p.name AS product, p.brand AS brand, p.price AS price;

// --> Products whose name begins with "Flex".


// ---------------------------------------------------------------- 4.9
// CONTAINS looks anywhere in the string.

MATCH (p:Product)
WHERE p.name CONTAINS "Watch"
RETURN p.name AS product, p.price AS price;


// ---------------------------------------------------------------- 4.10
// For case-INSENSITIVE matching, use a regular expression with =~
// The (?i) at the front means "ignore case".

MATCH (p:Product)
WHERE p.name =~ "(?i).*watch.*"
RETURN p.name AS product, p.price AS price;

// --> Now "watch", "Watch" and "WATCH" all match.


// ---------------------------------------------------------------- 4.11
// Missing data: IS NULL and IS NOT NULL.
// A property that was never set simply does not exist on that node.

MATCH (p:Product)
WHERE p.avgRating IS NULL
RETURN count(p) AS productsWithNoRating;


// ---------------------------------------------------------------- 4.12
// The reverse.

MATCH (p:Product)
WHERE p.avgRating IS NOT NULL AND p.avgRating >= 4.5
RETURN p.name AS product, p.avgRating AS rating
ORDER BY p.avgRating DESC
LIMIT 10;


// ---------------------------------------------------------------- 4.13
// Ranges: chain two comparisons with AND.

MATCH (p:Product)
WHERE p.price >= 100 AND p.price <= 300
RETURN p.name AS product, p.price AS price
ORDER BY p.price;

// --> 33 products between $100 and $300.


// ---------------------------------------------------------------- 4.14
// Booleans need no comparison — the property IS the condition.

MATCH (p:Product)
WHERE p.featured
RETURN p.name AS product, p.price AS price, p.featured AS featured
LIMIT 10;

// --> Same as "WHERE p.featured = true", but shorter and idiomatic.


// ---------------------------------------------------------------- 4.15
// Dates. Compare a datetime property against datetime("...").

MATCH (o:Order)
WHERE o.orderDate >= datetime("2024-01-01T00:00:00")
RETURN o.orderId AS orderId, o.orderDate AS placed, o.totalAmount AS amount
ORDER BY o.orderDate
LIMIT 10;


// ---------------------------------------------------------------- 4.16
// You can pull parts out of a date with dot notation.

MATCH (o:Order)
WHERE o.orderDate.year = 2024 AND o.orderDate.month = 6
RETURN o.orderId AS orderId, o.orderDate AS placed, o.status AS status
ORDER BY o.orderDate;

// --> June 2024 only. Also available: .day .hour .minute .quarter .weekday


// ---------------------------------------------------------------- 4.17
// Filter on a RELATIONSHIP property. Name the relationship first (r).

MATCH (o:Order)-[r:CONTAINS]->(p:Product)
WHERE r.quantity >= 3
RETURN o.orderId AS orderId, p.name AS product,
       r.quantity AS qty, r.totalPrice AS lineTotal
ORDER BY r.quantity DESC;

// --> Bulk lines. The filter is on the relationship, not on either node.


// ---------------------------------------------------------------- 4.18
// Filter on a PATTERN — "does this node have this relationship at all?"
// This is something SQL needs a subquery for.

MATCH (c:Customer)
WHERE (c)-[:WROTE]->(:Review)
RETURN c.name AS customer, c.segment AS segment;

// --> Only customers who have written at least one review.


// ---------------------------------------------------------------- 4.19
// NOT + pattern = "does NOT have this relationship".
// Extremely useful, and very hard to express in SQL.

MATCH (c:Customer)
WHERE NOT (c)-[:WROTE]->(:Review)
RETURN c.name AS customer, c.segment AS segment;

// --> Customers who have never reviewed anything.


// ---------------------------------------------------------------- 4.20
// Putting it together: valuable orders, placed on mobile, that were not cancelled.

MATCH (c:Customer)-[:PLACED]->(o:Order)
WHERE o.totalAmount > 2000
  AND o.deviceType = "mobile"
  AND o.status <> "Cancelled"
RETURN c.name        AS customer,
       c.segment     AS segment,
       o.orderId     AS orderId,
       o.totalAmount AS amount,
       o.status      AS status
ORDER BY o.totalAmount DESC;

//================================================================================
// SECTION 5 — RETRIEVING AND TRAVERSING
//================================================================================

// Filtering picks rows. Traversing follows relationships to reach data that is
// not on the node you started from. This is what a graph database is FOR.


// ---------------------------------------------------------------- 5.1
// DISTINCT removes duplicate rows.
// Without it, a customer appears once per order they placed.

MATCH (c:Customer)-[:PLACED]->(:Order)
RETURN c.name AS customer;

// --> 66 rows — one per order, so names repeat.


// ---------------------------------------------------------------- 5.2

MATCH (c:Customer)-[:PLACED]->(:Order)
RETURN DISTINCT c.name AS customer;

// --> 24 rows. One per customer.
//     Whenever a name repeats unexpectedly, the cause is almost always
//     that you traversed a relationship that fans out.


// ---------------------------------------------------------------- 5.3
// ORDER BY sorts. DESC for descending, ASC (the default) for ascending.

MATCH (p:Product)
RETURN p.name AS product, p.price AS price
ORDER BY p.price DESC
LIMIT 10;


// ---------------------------------------------------------------- 5.4
// Sort by two keys: brand alphabetically, then price high-to-low within brand.

MATCH (p:Product)
RETURN p.brand AS brand, p.name AS product, p.price AS price
ORDER BY p.brand ASC, p.price DESC
LIMIT 20;


// ---------------------------------------------------------------- 5.5
// SKIP + LIMIT = paging. This is page 2, at 10 rows per page.

MATCH (p:Product)
RETURN p.name AS product, p.price AS price
ORDER BY p.price DESC
SKIP 10 LIMIT 10;


// ---------------------------------------------------------------- 5.6
// ONE HOP: customer to their orders.

MATCH (c:Customer {customerId: 3068})-[:PLACED]->(o:Order)
RETURN c.name        AS customer,
       o.orderId     AS orderId,
       o.orderDate   AS placed,
       o.totalAmount AS amount,
       o.status      AS status
ORDER BY o.orderDate;

// --> Carol Glass's 3 orders.


// ---------------------------------------------------------------- 5.7
// TWO HOPS: customer to the products they bought, via their orders.
// The Order in the middle is anonymous — we don't need to name what we
// don't use.

MATCH (c:Customer {customerId: 3068})-[:PLACED]->(:Order)-[:CONTAINS]->(p:Product)
RETURN DISTINCT p.name AS product, p.brand AS brand, p.price AS price
ORDER BY p.price DESC;

// --> 15 distinct products.


// ---------------------------------------------------------------- 5.8
// THREE HOPS: customer -> order -> product -> category.

MATCH (c:Customer {customerId: 3068})-[:PLACED]->(:Order)
      -[:CONTAINS]->(p:Product)-[:IN_CATEGORY]->(cat:Category)
RETURN DISTINCT cat.name AS category, p.name AS product
ORDER BY category;

// --> Which categories this customer shops in.
//     Three relationships followed, in one readable line.


// ---------------------------------------------------------------- 5.9
// Keep the relationship so you can read its properties.

MATCH (c:Customer {customerId: 3068})-[:PLACED]->(o:Order)-[line:CONTAINS]->(p:Product)
RETURN o.orderId      AS orderId,
       p.name         AS product,
       line.quantity  AS qty,
       line.unitPrice AS unitPrice,
       line.totalPrice AS lineTotal
ORDER BY o.orderId, lineTotal DESC;


// ---------------------------------------------------------------- 5.10
// Two separate patterns, comma-separated. Both must match.
// Here: the customer's reviews AND which product each review is about.

MATCH (c:Customer)-[:WROTE]->(r:Review), (r)-[:REVIEWS]->(p:Product)
RETURN c.name    AS customer,
       p.name    AS product,
       r.rating  AS rating,
       r.title   AS title
ORDER BY r.rating
LIMIT 15;


// ---------------------------------------------------------------- 5.11
// The same thing written as one chained pattern. Usually clearer.

MATCH (c:Customer)-[:WROTE]->(r:Review)-[:REVIEWS]->(p:Product)
RETURN c.name AS customer, p.name AS product, r.rating AS rating
ORDER BY r.rating
LIMIT 15;


// ---------------------------------------------------------------- 5.12
// OPTIONAL MATCH is Cypher's LEFT JOIN. It keeps rows even when the
// pattern does not match, filling the missing values with null.
// Compare this with 5.13.

MATCH (c:Customer)
OPTIONAL MATCH (c)-[:WROTE]->(r:Review)
RETURN c.name AS customer, r.rating AS rating
ORDER BY customer
LIMIT 20;

// --> ALL 24 customers appear. Those with no reviews show null.


// ---------------------------------------------------------------- 5.13
// Plain MATCH drops them instead.

MATCH (c:Customer)-[:WROTE]->(r:Review)
RETURN c.name AS customer, r.rating AS rating
ORDER BY customer
LIMIT 20;

// --> Only customers who HAVE reviewed. The others vanish silently.
//     Choosing between MATCH and OPTIONAL MATCH is a real decision. Think
//     about whether "no match" should mean "no row" or "a row with nulls".


// ---------------------------------------------------------------- 5.14
// Variable-length paths. [:PARENT_OF*1..3] means "follow PARENT_OF
// between 1 and 3 times". This walks DOWN the category tree.

MATCH (top:Category {name: "Electronics"})-[:PARENT_OF*1..3]->(sub:Category)
RETURN sub.name AS subcategory, sub.level AS level
ORDER BY sub.level, sub.name;

// --> Everything beneath "Electronics", at any depth.
//     ALWAYS put an upper bound on the number of hops. An unbounded [*]
//     can try to walk your entire graph.


// ---------------------------------------------------------------- 5.15
// Walk UP instead, from a deep category to its root, and show the path.

MATCH path = (root:Category)-[:PARENT_OF*0..3]->(leaf:Category {name: "iPhone"})
WHERE root.level = 1
RETURN [n IN nodes(path) | n.name] AS breadcrumb, length(path) AS depth;

// --> ["Electronics", "Smartphones", "iPhone"]
//     [n IN nodes(path) | n.name] is a LIST COMPREHENSION: for each node on
//     the path, take its name.


// ---------------------------------------------------------------- 5.16
// THE CLASSIC GRAPH QUERY: "customers who bought the same thing I did".
// Four hops out and back. Note we exclude ourselves.

MATCH (me:Customer {customerId: 3068})-[:PLACED]->(:Order)-[:CONTAINS]->(p:Product)
      <-[:CONTAINS]-(:Order)<-[:PLACED]-(peer:Customer)
WHERE peer <> me
RETURN DISTINCT peer.name AS alsoBought, p.name AS sharedProduct
ORDER BY alsoBought;

// --> Customers who share a purchase with Carol Glass.
//     Read the pattern as a shape: out to a product, then back in from
//     another direction. That "there and back" motion is the heart of
//     collaborative filtering.


// ---------------------------------------------------------------- 5.17
// One more hop turns that into a RECOMMENDATION: what did those peers buy
// that I have not?

MATCH (me:Customer {customerId: 3068})-[:PLACED]->(:Order)-[:CONTAINS]->(shared:Product)
      <-[:CONTAINS]-(:Order)<-[:PLACED]-(peer:Customer)
      -[:PLACED]->(:Order)-[:CONTAINS]->(rec:Product)
WHERE peer <> me
  AND NOT (me)-[:PLACED]->(:Order)-[:CONTAINS]->(rec)
RETURN DISTINCT rec.name AS recommendation, rec.brand AS brand, rec.price AS price
ORDER BY recommendation;

// --> A recommendation engine. Six relationships, one query, no joins.
//     In SQL this is six JOINs over three tables, and it gets slower with
//     every hop. Here the cost only depends on how many neighbours you touch.


// ---------------------------------------------------------------- 5.18
// collect() gathers many rows into a single list — the opposite of UNWIND.

MATCH (c:Customer)-[:PLACED]->(:Order)-[:CONTAINS]->(p:Product)
RETURN c.name AS customer, collect(DISTINCT p.name) AS productsBought
ORDER BY customer
LIMIT 10;

// --> One row per customer, with a LIST of their products.
//     Much easier to read than 229 separate rows.

//================================================================================
// SECTION 6 — AGGREGATION
//================================================================================

// Aggregation turns many rows into one summary row: counts, sums, averages.

// THE ONE THING YOU MUST UNDERSTAND:
// Cypher has no GROUP BY. Grouping is IMPLICIT. Every non-aggregate expression
// in your RETURN becomes a grouping key, automatically.

//    RETURN count(*)                    // one row for everything
//    RETURN c.segment, count(*)         // one row PER SEGMENT

// If you get more rows than expected, you have an extra non-aggregate column
// in your RETURN. That is the cause, essentially every time.


// ---------------------------------------------------------------- 6.1
// count(*) counts rows.

MATCH (c:Customer)
RETURN count(*) AS totalCustomers;

// --> 24


// ---------------------------------------------------------------- 6.2
// count(*) counts ROWS. count(x) counts NON-NULL VALUES of x.
// The difference shows up with OPTIONAL MATCH.

MATCH (c:Customer)
OPTIONAL MATCH (c)-[:WROTE]->(r:Review)
RETURN count(*)  AS rows,
       count(r)  AS actualReviews,
       count(DISTINCT c) AS customers;

// --> rows > actualReviews, because customers with no reviews produce a
//     row where r is null. count(r) skips those nulls; count(*) does not.


// ---------------------------------------------------------------- 6.3
// DISTINCT inside an aggregate counts unique values only.

MATCH (p:Product)
RETURN count(*)                 AS products,
       count(DISTINCT p.brand)  AS distinctBrands,
       count(DISTINCT p.category) AS distinctCategories;

// --> 186 products, 84 brands, 44 categories.


// ---------------------------------------------------------------- 6.4
// IMPLICIT GROUPING. Add a non-aggregate column and you get one row per
// distinct value of it. No GROUP BY clause is needed — or allowed.

MATCH (o:Order)
RETURN o.status AS status, count(*) AS orders
ORDER BY orders DESC;

// --> Delivered 25, Shipped 19, Cancelled 9, Returned 9, Processing 4
//     o.status is not an aggregate, so it became the grouping key.


// ---------------------------------------------------------------- 6.5
// Same idea on customers.

MATCH (c:Customer)
RETURN c.segment AS segment, count(*) AS customers
ORDER BY customers DESC;

// --> VIP 18, At-Risk 2, Churned 2, New 2


// ---------------------------------------------------------------- 6.6
// sum() adds numbers up.

MATCH (o:Order)
WHERE o.status IN ["Delivered", "Shipped"]
RETURN count(*)             AS orders,
       sum(o.totalAmount)   AS totalRevenue;

// --> 44 orders, 167129.79


// ---------------------------------------------------------------- 6.7
// avg(), min() and max(). round() tidies the decimals.

MATCH (o:Order)
RETURN round(avg(o.totalAmount), 2) AS avgOrderValue,
       min(o.totalAmount)           AS smallestOrder,
       max(o.totalAmount)           AS largestOrder,
       count(*)                     AS orders;

// --> avg 3457.69, min 16.86, max 13502.87, over 66 orders.


// ---------------------------------------------------------------- 6.8
// Several aggregates, grouped by one key. This is the shape of most
// real analytics queries you will ever write.

MATCH (c:Customer)-[:PLACED]->(o:Order)
RETURN c.segment                    AS segment,
       count(*)                     AS orders,
       round(sum(o.totalAmount), 2) AS revenue,
       round(avg(o.totalAmount), 2) AS avgOrderValue
ORDER BY revenue DESC;

// --> VIP 54 orders, avg 3844.32
//     New 4 orders, avg 436.38
//     New customers spend far less per order. That is a real finding from
//     your data, produced by one query.


// ---------------------------------------------------------------- 6.9
// Aggregate over a relationship property. Which products earn the most?

MATCH (:Order)-[line:CONTAINS]->(p:Product)
RETURN p.name                       AS product,
       count(*)                     AS timesOrdered,
       sum(line.quantity)           AS unitsSold,
       round(sum(line.totalPrice), 2) AS revenue
ORDER BY revenue DESC
LIMIT 10;

// --> Top earner: Flex Sofa at 12313.44


// ---------------------------------------------------------------- 6.10
// Which customers spend the most? Note we count DISTINCT orders — without
// DISTINCT, an order with 3 lines would be counted three times.

MATCH (c:Customer)-[:PLACED]->(o:Order)-[line:CONTAINS]->(:Product)
RETURN c.name                        AS customer,
       c.segment                     AS segment,
       count(DISTINCT o)             AS orders,
       round(sum(line.totalPrice), 2) AS spend
ORDER BY spend DESC
LIMIT 10;

// --> Top spender: Jessica Johnson at 24338.54
//     The DISTINCT on orders matters. Take it out and compare.


// ---------------------------------------------------------------- 6.11
// Average rating per product, from the Review nodes.

MATCH (r:Review)-[:REVIEWS]->(p:Product)
RETURN p.name                  AS product,
       count(r)                AS reviews,
       round(avg(r.rating), 2) AS avgRating
ORDER BY avgRating DESC, reviews DESC
LIMIT 15;


// ---------------------------------------------------------------- 6.12
// Overall review statistics.

MATCH (r:Review)
RETURN count(*)                AS totalReviews,
       round(avg(r.rating), 2) AS avgRating,
       min(r.rating)           AS worst,
       max(r.rating)           AS best;

// --> 63 reviews, average 4.03, from 1 to 5.


// ---------------------------------------------------------------- 6.13
// Distribution: how many reviews at each star rating?

MATCH (r:Review)
RETURN r.rating AS stars, count(*) AS reviews
ORDER BY stars DESC;

// --> 5 stars: 25, 4 stars: 24, 3 stars: 7, 2 stars: 5, 1 stars: 2


// ---------------------------------------------------------------- 6.14
// WITH is the key to multi-stage queries. It pipes the result of one part
// into the next, and it is where you filter on an AGGREGATE.
// (In SQL this would be HAVING. Cypher has no HAVING — it has WITH.)

MATCH (c:Customer)-[:PLACED]->(o:Order)
WITH c, count(*) AS orders, sum(o.totalAmount) AS spend
WHERE orders >= 3
RETURN c.name           AS customer,
       orders,
       round(spend, 2)  AS spend
ORDER BY spend DESC;

// --> Only customers with 3 or more orders.
//     WHERE after WITH filters on the aggregate. WHERE after MATCH cannot —
//     the aggregate does not exist yet at that point.


// ---------------------------------------------------------------- 6.15
// Aggregate, then keep traversing. This is what WITH really unlocks.
// Find the top 5 spenders, THEN go and fetch what they bought.

MATCH (c:Customer)-[:PLACED]->(o:Order)
WITH c, sum(o.totalAmount) AS spend
ORDER BY spend DESC
LIMIT 5
MATCH (c)-[:PLACED]->(:Order)-[:CONTAINS]->(p:Product)
RETURN c.name                    AS customer,
       round(spend, 2)           AS spend,
       count(DISTINCT p)         AS distinctProducts,
       collect(DISTINCT p.brand)[0..5] AS someBrands
ORDER BY spend DESC;

// --> The aggregate is computed first, then used as a starting point for
//     more traversal. You cannot do this in a single SQL SELECT.


// ---------------------------------------------------------------- 6.16
// Group by two keys at once — just list both non-aggregate columns.

MATCH (c:Customer)-[:PLACED]->(o:Order)
RETURN c.segment  AS segment,
       o.status   AS status,
       count(*)   AS orders
ORDER BY segment, orders DESC;

// --> One row per (segment, status) combination.


// ---------------------------------------------------------------- 6.17
// Orders over time. Group by year and month.

MATCH (o:Order)
RETURN o.orderDate.year  AS year,
       o.orderDate.month AS month,
       count(*)          AS orders,
       round(sum(o.totalAmount), 2) AS revenue
ORDER BY year, month;

// --> A monthly trend, straight out of the datetime property.


// ---------------------------------------------------------------- 6.18
// Percentage of total. Compute the total FIRST with WITH, then use it.

MATCH (o:Order)
WITH count(*) AS grandTotal
MATCH (o:Order)
RETURN o.status AS status,
       count(*) AS orders,
       round(100.0 * count(*) / grandTotal, 1) AS percentage
ORDER BY orders DESC;

// --> Percentages summing to 100. Note 100.0 not 100 — integer division
//     would round everything down to 0.


// ---------------------------------------------------------------- 6.19
// A count subquery: count { } counts matches WITHOUT creating extra rows.
// Compare this to the alternative below.

MATCH (p:Product)
WHERE p.price > 500
RETURN p.name AS product,
       p.price AS price,
       count { (p)<-[:CONTAINS]-(:Order) }   AS timesOrdered,
       count { (p)<-[:REVIEWS]-(:Review) }   AS reviewCount
ORDER BY timesOrdered DESC
LIMIT 15;

// --> Two independent counts on one row. Doing this with two MATCH clauses
//     would multiply the rows together and give wrong numbers.


// ---------------------------------------------------------------- 6.20
// Putting everything together: a category performance report.

MATCH (cat:Category)<-[:IN_CATEGORY]-(p:Product)<-[line:CONTAINS]-(o:Order)
WHERE o.status IN ["Delivered", "Shipped"]
WITH cat,
     count(DISTINCT p)              AS products,
     count(DISTINCT o)              AS orders,
     sum(line.quantity)             AS unitsSold,
     round(sum(line.totalPrice), 2) AS revenue
WHERE revenue > 500
RETURN cat.name AS category, products, orders, unitsSold, revenue,
       round(revenue / orders, 2) AS revenuePerOrder
ORDER BY revenue DESC
LIMIT 15;

// --> A genuine business report: five aggregates, a filter on one of them,
//     and a derived metric. If you can read this query, you can read Cypher.

// READING
//   MATCH (n:Label)                      find nodes
//   MATCH (n:Label {prop: value})        find with inline equality filter
//   MATCH (a)-[:REL]->(b)                follow a relationship
//   MATCH (a)-[:REL]-(b)                 either direction
//   MATCH (a)-[r:REL]->(b)               name it to read r.property
//   MATCH (a)-[:REL*1..3]->(b)           1 to 3 hops (always bound it)
//   MATCH path = (a)-[:REL]->(b)         capture the whole path
//   OPTIONAL MATCH ...                   keep rows even when it doesn't match
//   RETURN x AS name                     name your columns
//   ORDER BY x DESC / SKIP n / LIMIT n   sort and page
//   DISTINCT                             remove duplicate rows

// FILTERING
//   WHERE n.p = 1                        = <> < > <= >=
//   WHERE n.p IN [1, 2, 3]               membership
//   WHERE n.name STARTS WITH "A"         also ENDS WITH, CONTAINS
//   WHERE n.name =~ "(?i).*abc.*"        regex, (?i) = ignore case
//   WHERE n.p IS NULL / IS NOT NULL      missing values
//   WHERE (a)-[:REL]->(b)                pattern as a true/false test
//   WHERE NOT (a)-[:REL]->(b)            negated pattern
//   WHERE n.date.year = 2024             .year .month .day .hour .quarter

// WRITING
//   CREATE (n:Label {p: 1})              always creates (can duplicate)
//   MERGE  (n:Label {key: 1})            find-or-create (safe to re-run)
//   SET n.p = 1                          add or update a property
//   SET n:NewLabel                       add a label
//   REMOVE n.p / REMOVE n:Label          take away
//   DELETE r                             delete a relationship
//   DETACH DELETE n                      delete a node and its relationships
//   UNWIND [...] AS row                  turn a list into rows

// AGGREGATING  (grouping is implicit — no GROUP BY exists)
//   count(*)                             count rows
//   count(x) / count(DISTINCT x)         count non-null / unique values
//   sum() avg() min() max()              the usual
//   round(x, 2)                          tidy decimals
//   collect(x)                           gather rows into a list
//   count { (n)-[:REL]->() }             count without fanning out rows
//   WITH ... WHERE ...                   this is HAVING

// INSPECTING
//   labels(n) / type(r) / keys(n)        what is this?
//   properties(n)                        all properties as a map
//   nodes(path) / relationships(path)    what is on this path?
//   length(path)                         number of RELATIONSHIPS
//   SHOW CONSTRAINTS / SHOW INDEXES      what is defined?

// RESET
//   MATCH (n) DETACH DELETE n            empty the whole database
